# PowerShell LNK Extractor Script

$showPaths = $true

# Collect all filesystem roots
$drives = @()

# Standard drives
$drives += Get-PSDrive -PSProvider FileSystem |
    Select-Object -ExpandProperty Root

# Mounted volumes without drive letters
$volumeGuids = mountvol.exe |
    Select-String '\\\?\\Volume\{.*\}\\' |
    ForEach-Object { $_.Matches.Value }

$drives += $volumeGuids
$drives = $drives | Sort-Object -Unique

$wsh = New-Object -ComObject WScript.Shell

$results = foreach ($drive in $drives) {

    Write-Host "[*] Scanning $drive"

    Get-ChildItem `
        -LiteralPath $drive `
        -Filter "*.lnk" `
        -File `
        -Force `
        -Recurse `
        -ErrorAction SilentlyContinue |

    ForEach-Object {

        try {

            $shortcut = $wsh.CreateShortcut($_.FullName)

            [PSCustomObject]@{
                ScanRoot       = $drive
                Name           = $_.Name
                FullPath       = $_.FullName
                Directory      = $_.DirectoryName
                SizeKB         = [math]::Round($_.Length / 1KB, 2)

                CreationTime   = $_.CreationTimeUtc
                LastAccessTime = $_.LastAccessTimeUtc
                LastWriteTime  = $_.LastWriteTimeUtc

                Attributes     = $_.Attributes.ToString()

                TargetPath     = $shortcut.TargetPath
                Arguments      = $shortcut.Arguments
                WorkingDir     = $shortcut.WorkingDirectory
                Description    = $shortcut.Description
                IconLocation   = $shortcut.IconLocation
                Hotkey         = $shortcut.Hotkey
                WindowStyle    = $shortcut.WindowStyle
            }

        } catch {

            [PSCustomObject]@{
                ScanRoot = $drive
                Name     = $_.Name
                FullPath = $_.FullName
                Error    = "Failed to parse shortcut"
            }
        }
    }
}

# Remove duplicates
$results = $results | Sort-Object FullPath -Unique


# =========================
# JSON OUTPUT
# =========================
$outJson = ".\out.json"
$results | ConvertTo-Json -Depth 6 | Set-Content $outJson -Encoding UTF8
Write-Host "[+] JSON saved to $outJson"
